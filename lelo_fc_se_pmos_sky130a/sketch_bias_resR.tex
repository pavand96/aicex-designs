% Resistor-degenerated wide-swing cascode bias - sky130A LVT, VDD = 0.9 V.
%
% Architecture (matches user-sketched topology):
%
%   Col 1 (master): IBIAS pin -> PMOS diode -> VBP
%                                              (10 uA forced from external pin)
%
%   Col 2 (NMOS bias rail): VBP gates a PMOS source (10 uA mirror)
%                            into VBIAS2 node, then R_n (10 kohm, drops Vov)
%                            into VBIAS3, then NMOS diode replica to VSS.
%
%       VBIAS3 = V_T  + V_ov      (= gate of signal-path NMOS bottom XM11/XM12)
%       VBIAS2 = VBIAS3 + V_ov    (= gate of signal-path NMOS cascode XM1A/XM2A)
%
%   Col 3 (PMOS bias rail): R_p (10 kohm) drops Vov from VBP to VBIAS1,
%                            current sunk by NMOS replica (gate=VBIAS3) to VSS.
%
%       VBIAS1 = VBP - V_ov       (= gate of signal-path PMOS cascode XM3A/XM4A)
%
%   Signal path (right block, unchanged): PMOS tail (g=VBP), PMOS input
%   pair, fold node fL with PMOS load mirror (XM3 diode) -> single-ended VOUTP.
%
% No external pin removed (IBIAS retained); R_n, R_p are on-chip poly drop
% resistors that synthesise the Vov offset between source-rail and cascode-rail.
% Single DC solution per leg (no bootstrap): NMOS leg sourced from VBP,
% PMOS leg pulled by mirror referenced to VBIAS3.

\documentclass[border=10pt]{standalone}
\usepackage{amsmath}
\usepackage[siunitx,europeanresistors]{circuitikz}
\usetikzlibrary{positioning,calc}

\ctikzset{
  tripoles/mos style=arrows,
  transistors/arrow pos=end,
  nodes width=0.06,
  font=\small,
}

\begin{document}
\begin{circuitikz}[scale=0.9, transform shape]

% =========================================================================
%  RAIL PLANES
% =========================================================================
\draw[thick] (-1,10) -- (17,10) node[above,pos=0.02]{$V_{DD}=0.9\,\mathrm{V}$};
\draw[thick] (-1,-0.5) -- (17,-0.5) node[below,pos=0.02]{$V_{SS}$};

% =========================================================================
%  COL 1  --  MASTER PMOS DIODE AT IBIAS PIN
% =========================================================================
\node at (0.5,10.4) {\small Col 1: master};
\draw (0.5,10) to[short,*-] (0.5,9);
\draw (0.5,9) node[pmos,xscale=-1] (MREF) {};
\node[right=2pt of MREF.gate] {\scriptsize $M_{\mathrm{REF}}$};
\node[left=2pt of MREF.gate]  {\scriptsize $\tfrac{16\times 4\times 4}{1}$};
% gate-drain short (diode)
\draw (MREF.drain) to[short,*-] (MREF.drain |- 0,7);
\draw (MREF.drain |- 0,7) to[short,-*] (MREF.gate);
\node[circle,fill=black,inner sep=1pt] at (MREF.gate) {};
\draw (MREF.gate) -- ++(1.4,0) node[above]{$V_{BP}$} -- ++(0.6,0) coordinate (vbp_bus);
\draw (MREF.drain) to[R=$R_{BP}$,*-*] (MREF.drain |- 0,6) coordinate (ibias_pin);
% external IBIAS pin annotation
\draw (ibias_pin) -- ++(-0.8,0) node[left]{\footnotesize IBIAS pin};
\draw (ibias_pin) -- ++(0,-0.6) node[below,align=center]{\scriptsize ext. current\\\scriptsize sink $I_{REF}=10\,\mu$A};

% =========================================================================
%  VBP HORIZONTAL BUS (carries gate-bias to Col 2 source mirror + signal tail)
% =========================================================================
\draw[blue!60!black,thick,dashed] (vbp_bus) -- (16,8.6);
\node[blue!60!black] at (15.5,8.4) {\footnotesize $V_{BP}$ bus};

% =========================================================================
%  COL 2 -- NMOS bias rail leg
%   VBP -> PMOS source -> VBIAS2 -> R_n -> VBIAS3 -> NMOS diode -> VSS
% =========================================================================
\node at (4.5,10.4) {\small Col 2: NMOS bias leg};
\draw (4.5,10) -- (4.5,9);
\draw (4.5,9) node[pmos,xscale=-1] (MPNLEG) {};
\node[right=2pt of MPNLEG.gate] {\scriptsize $M_{P,NL}$};
% gate of MPNLEG taps the VBP bus
\draw (MPNLEG.gate) -- (vbp_bus |- MPNLEG.gate);
\node[circle,fill=blue!60!black,inner sep=1pt] at (vbp_bus |- MPNLEG.gate) {};
% drain -> VBIAS2 node
\draw (MPNLEG.drain) -- (MPNLEG.drain |- 0,7.5) coordinate (vb2);
\node[circle,fill=black,inner sep=1pt] at (vb2) {};
\draw (vb2) -- ++(1.6,0) node[above]{$V_{BIAS2}$} -- ++(0.6,0) coordinate (vb2_bus);
% R_n
\draw (vb2) to[R,l=$R_n$,*-*] (vb2 |- 0,5.5) coordinate (vb3);
\draw (vb3) -- ++(1.6,0) node[above]{$V_{BIAS3}$} -- ++(0.6,0) coordinate (vb3_bus);
% NMOS diode bottom
\draw (vb3) -- (vb3 |- 0,4) coordinate (vb3d);
\draw (vb3d) node[nmos] (MNNDIO) {};
\node[right=2pt of MNNDIO.gate] {\scriptsize $M_{N,DIO}$};
\node[left=2pt of MNNDIO.gate]  {\scriptsize $\tfrac{4\times 2\times 4}{0.5}$};
% diode tie: gate-drain short
\draw (MNNDIO.drain) to[short,*-] (MNNDIO.drain |- 0,3.2);
\draw (MNNDIO.drain |- 0,3.2) -| (MNNDIO.gate);
\node[circle,fill=black,inner sep=1pt] at (MNNDIO.gate) {};
\draw (MNNDIO.source) -- (4.5,-0.5);

% =========================================================================
%  COL 3 -- PMOS bias rail leg
%   VBP -> R_p -> VBIAS1 -> NMOS replica sink (g=VBIAS3) -> VSS
% =========================================================================
\node at (8.5,10.4) {\small Col 3: PMOS bias leg};
% take VBP off bus
\draw[blue!60!black,thick,dashed] (vbp_bus |- 0,9) -- (8.5,9);
\node[circle,fill=blue!60!black,inner sep=1pt] at (vbp_bus |- 0,9) {};
\draw (8.5,9) to[short,*-] (8.5,8) coordinate (rp_top);
% R_p
\draw (rp_top) to[R,l=$R_p$,*-*] (rp_top |- 0,6) coordinate (vb1);
\draw (vb1) -- ++(1.6,0) node[above]{$V_{BIAS1}$} -- ++(0.6,0) coordinate (vb1_bus);
% NMOS sink
\draw (vb1) -- (vb1 |- 0,4.5) coordinate (mpsd);
\draw (mpsd) node[nmos] (MNPSNK) {};
\node[right=2pt of MNPSNK.gate] {\scriptsize $M_{N,PSNK}$};
\node[left=2pt of MNPSNK.gate]  {\scriptsize $\tfrac{4\times 2\times 4}{0.5}$};
% gate taps VBIAS3 bus
\draw (MNPSNK.gate) -- (vb3_bus |- MNPSNK.gate);
\node[circle,fill=black,inner sep=1pt] at (vb3_bus |- MNPSNK.gate) {};
\draw (MNPSNK.source) -- (8.5,-0.5);

% =========================================================================
%  BIAS BUSES TO RIGHT BLOCK (signal path)
% =========================================================================
\node[circle,fill=blue!60!black,inner sep=1pt] at (16,8.6) {};
\draw[blue!60!black,thick,dashed] (vb1_bus) -- (16,6);   \node[blue!60!black] at (15.5,5.8) {\footnotesize $V_{BIAS1}$ bus};
\draw[black,thick,dashed]         (vb2_bus) -- (16,7.5); \node at (15.5,7.3) {\footnotesize $V_{BIAS2}$ bus};
\draw[black,thick,dashed]         (vb3_bus) -- (16,5.5); \node at (15.5,5.3) {\footnotesize $V_{BIAS3}$ bus};

% =========================================================================
%  SIGNAL PATH (compact symbolic representation on right)
% =========================================================================
\node at (12.8,10.4) {\small Folded-cascode signal path (unchanged)};
% Symbolic block - just labels
\node[draw,rounded corners,minimum width=4.5cm,minimum height=8.5cm,fill=black!3] at (12.8,4.8) (SIG) {};
\node[align=center] at (12.8,8.6) {\footnotesize PMOS tail $M_{TL}$ \\ \footnotesize (g\,=\,$V_{BP}$)};
\node[align=center] at (12.8,7.5) {\footnotesize PMOS pair $M_1$/$M_2$ \\ \footnotesize $V_{in+}$/$V_{in-}$};
\node[align=center] at (12.8,6.4) {\footnotesize PMOS cascodes $M_{3A}$/$M_{4A}$ \\ \footnotesize (g\,=\,$V_{BIAS1}$)};
\node[align=center] at (12.8,5.2) {\footnotesize PMOS diode load $M_3$ @ $f_L$ \\ \footnotesize $M_4$ mirror $\to V_{OUTP}$};
\node[align=center] at (12.8,4.0) {\footnotesize NMOS cascodes $M_{1A}$/$M_{2A}$ \\ \footnotesize (g\,=\,$V_{BIAS2}$)};
\node[align=center] at (12.8,2.7) {\footnotesize NMOS bottom $M_{11}$/$M_{12}$ \\ \footnotesize (g\,=\,$V_{BIAS3}$)};
\node[align=center] at (12.8,1.3) {\footnotesize $\to V_{OUTP}$};

% =========================================================================
%  TITLE
% =========================================================================
\node at (8,11.2) {\bfseries Wide-swing cascode bias (R-degenerated) -- sky130A LVT, VDD=0.9\,V};
